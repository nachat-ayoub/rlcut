cmake_minimum_required(VERSION 3.16)
project(rlcut_ai_gpu LANGUAGES C CXX)
option(RLCUT_AI_VULKAN "Build the NCNN worker with Vulkan acceleration" ON)
set(NCNN_VULKAN ${RLCUT_AI_VULKAN} CACHE BOOL "" FORCE)
set(NCNN_SYSTEM_GLSLANG OFF CACHE BOOL "" FORCE)
set(NCNN_BUILD_TOOLS OFF CACHE BOOL "" FORCE)
set(NCNN_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(NCNN_BUILD_BENCHMARK OFF CACHE BOOL "" FORCE)
set(NCNN_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(NCNN_SHARED_LIB OFF CACHE BOOL "" FORCE)
add_subdirectory("${NCNN_SOURCE}" ncnn)
find_package(PNG REQUIRED)
add_executable(rlcut-ai-gpu ../ai-cpu.cpp)
# ncnn exports -fno-exceptions; the worker protocol catches errors per request.
if(MSVC)
  set_source_files_properties(../ai-cpu.cpp PROPERTIES COMPILE_OPTIONS /EHsc)
else()
  set_source_files_properties(../ai-cpu.cpp PROPERTIES COMPILE_OPTIONS -fexceptions)
endif()
target_compile_features(rlcut-ai-gpu PRIVATE cxx_std_17)
target_link_libraries(rlcut-ai-gpu PRIVATE ncnn PNG::PNG)
